d=json.load(sys.stdin)
markets = {}
for e in d['liveEvents']:
    for m in e.get('fullEvent', {}).get('rawOdds', []):
        mid = m.get('marketId')
        hc = m.get('handicap')
        outs = [o.get('name') for o in m.get('outcomes', [])]
        key = (mid, tuple(outs))
        if key not in markets:
            markets[key] = {'handicap': hc, 'count': 0}
        markets[key]['count'] += 1

for (mid, outs), info in sorted(markets.items(), key=lambda x: -x[1]['count'])[:20]:
    print(f'MarketId: {mid}, Outcomes: {outs}, Handicap: {info[\"handicap\"]}, Count: {info[\"count\"]}')
